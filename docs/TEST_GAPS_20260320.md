## 缺口 2：text_processor 多執行緒 race condition（ATK-003）

**問題**：`replace_rules` 全域快取在多執行緒下互相污染。

**建議測試函式**：

```python
def test_text_processor_different_rules_do_not_pollute_each_other():
    """兩個執行緒使用不同 rules，翻譯結果不應互相污染。"""
    import threading
    
    rules_A = [{"match": "hello", "replace": "A_RESULT"}]
    rules_B = [{"match": "hello", "replace": "B_RESULT"}]
    results = {}
    
    def translate_with_rules(rules, label):
        load_rules(rules)  # 或傳入翻譯函式
        result = apply_replace_rules("hello world", rules)
        results[label] = result
    
    t1 = threading.Thread(target=translate_with_rules, args=(rules_A, "A"))
    t2 = threading.Thread(target=translate_with_rules, args=(rules_B, "B"))
    t1.start(); t2.start()
    t1.join(); t2.join()
    
    assert results["A"] == "A_RESULT world", f"A 吃到錯誤規則：{results['A']}"
    assert results["B"] == "B_RESULT world", f"B 吃到錯誤規則：{results['B']}"

def test_text_processor_concurrent_initialization():
    """10 個執行緒同時初始化不同 rules，最終各自結果正確。"""
    # 大量並發測試
    import threading
    results = {}
    
    def worker(i):
        rules = [{"match": f"TOKEN_{i}", "replace": f"RESULT_{i}"}]
        load_rules(rules)
        results[i] = apply_replace_rules(f"TOKEN_{i}", rules)
    
    threads = [threading.Thread(target=worker, args=(i,)) for i in range(10)]
    for t in threads: t.start()
    for t in threads: t.join()
    
    for i in range(10):
        assert results[i] == f"RESULT_{i}"
```

def test_highly_compressed_zip_raises_exception():
    """壓縮比超過閾值（如 1000x）的 ZIP 應拋出 CompressionBombError。"""
    import zipfile, io, pytest
    
    # 建立一個極度壓縮的 ZIP（1KB 內容壓縮成 100 bytes）
    content = b'X' * (1024 * 1024)  # 1MB data -> 100 bytes compressed
    zf = zipfile.ZipFile(io.BytesIO(), 'w', compression=zipfile.ZIP_DEFLATED)
    zf.writestr("large_file.txt", content)
    
    # 嘗試讀取
    result_zf = zipfile.ZipFile(io.BytesIO(zf.fp.getvalue()))
    
    with pytest.raises((CompressionBombError, ValueError)):
        read_text_from_zip(result_zf, "large_file.txt", max_expansion_ratio=1000)

def test_legitimate_zip_with_normal_ratio_succeeds():
    """正常壓縮比（< 100x）的 ZIP 應正常處理。"""
    import zipfile, io
    
    content = b'This is normal content' * 100  # ~1.7KB
    zf = zipfile.ZipFile(io.BytesIO(), 'w')
    zf.writestr("normal.txt", content)
    
    result = read_text_from_zip(zf, "normal.txt")

def test_concurrent_api_key_rotation_no_duplicate_keys():
    """10 個執行緒同時輪替 API key，確認 key 均勻分布。"""
    import threading
    
    allocated_keys = []
    lock = threading.Lock()
    
    def get_and_record():
        key = get_current_api_key()  # 應輪替到不同 key

    unique_keys = set(allocated_keys)

## 缺口 9：SQLite WAL Crash Recovery（ATK-011）

**問題**：`synchronous=OFF` 在 daemon crash 後 WAL 無法正確 recovery。

**建議測試函式**：

```python
def test_cache_search_db_after_unclean_shutdown():
    """模拟 unclean shutdown（WAL 未 checkpoint），確認 DB 可正常開啟。"""
    import sqlite3, os
    
    # 建立 DB 並寫入
    conn = sqlite3.connect(test_db_path, check_same_thread=False)
    conn.execute("PRAGMA journal_mode = WAL")
    conn.execute("PRAGMA synchronous = OFF")  # 模擬現有設定
    conn.execute("CREATE TABLE test(id INTEGER PRIMARY KEY)")
    conn.execute("INSERT INTO test VALUES (1)")
    conn.commit()
    conn.close()
    
    # 模擬 unclean shutdown（WAL 未 checkpoint）
    # 直接刪除 -wal 和 -shm 檔案
    for suffix in ['-wal', '-shm']:
        p = Path(str(test_db_path) + suffix)
        if p.exists(): p.unlink()
    
    # 重新開啟，應可正常運作（不做 WAL recovery 或做並成功）
    conn2 = sqlite3.connect(test_db_path)
    result = conn2.execute("SELECT * FROM test").fetchone()
    assert result == (1,), "資料應存在"
```
